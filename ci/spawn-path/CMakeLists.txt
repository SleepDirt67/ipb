cmake_minimum_required(VERSION 3.20)
project(ixray_spawn_path_capture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(ixray_stack_extract capture/stack_extract.cpp)
target_link_libraries(ixray_stack_extract PRIVATE advapi32 dbghelp)
add_executable(ixray_etw_decode capture/etw_decode.cpp)
target_link_libraries(ixray_etw_decode PRIVATE advapi32)
add_executable(ixray_display_path capture/display_extract.cpp)
target_link_libraries(ixray_display_path PRIVATE advapi32)
add_executable(spawn_profile_fixture fixture.cpp)
target_compile_definitions(spawn_profile_fixture PRIVATE NOMINMAX)
target_include_directories(spawn_profile_fixture PRIVATE "${ENGINE}/sdk/include" "${ENGINE}/src/xrScripts")
target_link_libraries(spawn_profile_fixture PRIVATE "${LUA_LIB}")
