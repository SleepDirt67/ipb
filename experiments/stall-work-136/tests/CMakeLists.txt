cmake_minimum_required(VERSION 3.21)
project(IXRaySemanticTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
foreach(variable GENERATED ENGINE LUA_LIB)
  file(TO_CMAKE_PATH "${${variable}}" ${variable})
endforeach()
add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
if(MSVC)
  add_compile_options(/utf-8)
endif()
foreach(name filesystem regex lua_lookup ini ini_evaluation)
  add_executable(${name} "${GENERATED}/${name}.cpp")
endforeach()
target_include_directories(lua_lookup PRIVATE "${ENGINE}/sdk/include/lua")
target_link_libraries(lua_lookup PRIVATE "${LUA_LIB}")
